 SELECT distinct proveedor AS proveedor,
    min(a.precio) as "Precio mínimo",
	max(a.precio) as "Precio máximo",
	avg(a.precio) as "Precio promedio",
    sum(a.cantidad) as "Cantidad total"
   FROM electrodomesticos.vista_proveedores_productos as a
   group by proveedor;
