-- Table: empleos.usuarios

DROP TABLE IF EXISTS empleos.usuarios;

create sequence empleos.usuarios_usuario_id_seq;

CREATE TABLE IF NOT EXISTS empleos.usuarios
(
    usuario_id integer NOT NULL DEFAULT nextval('empleos.usuarios_usuario_id_seq'::regclass),
    nombre character varying(100) COLLATE pg_catalog."default" NOT NULL,
    apellido character varying(100) COLLATE pg_catalog."default" NOT NULL,
    correo_electronico character varying(255) COLLATE pg_catalog."default" NOT NULL,
    password_hash character varying(255) COLLATE pg_catalog."default" NOT NULL,
    fecha_registro timestamp with time zone DEFAULT CURRENT_TIMESTAMP,
    telefono character varying(50) COLLATE pg_catalog."default",
    direccion text COLLATE pg_catalog."default",
    CONSTRAINT usuarios_pkey PRIMARY KEY (usuario_id),
    CONSTRAINT uk_correo_electronico UNIQUE (correo_electronico)
)

TABLESPACE pg_default;

ALTER TABLE IF EXISTS empleos.usuarios
    OWNER to postgres;


-- Table: empleos.experiencia_laboral

create sequence empleos.experiencia_laboral_experiencia_id_seq;

DROP TABLE IF EXISTS empleos.experiencia_laboral;

CREATE TABLE IF NOT EXISTS empleos.experiencia_laboral
(
    experiencia_id integer NOT NULL DEFAULT nextval('empleos.experiencia_laboral_experiencia_id_seq'::regclass),
    usuario_id integer NOT NULL,
    nombre_empresa character varying(255) COLLATE pg_catalog."default" NOT NULL,
    cargo character varying(255) COLLATE pg_catalog."default" NOT NULL,
    fecha_inicio date NOT NULL,
    fecha_fin date,
    descripcion text COLLATE pg_catalog."default",
    CONSTRAINT experiencia_laboral_pkey PRIMARY KEY (experiencia_id),
    CONSTRAINT fk_usuario_id FOREIGN KEY (usuario_id)
        REFERENCES empleos.usuarios (usuario_id) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE CASCADE
)

TABLESPACE pg_default;

ALTER TABLE IF EXISTS empleos.experiencia_laboral
    OWNER to postgres;

COMMENT ON CONSTRAINT fk_usuario_id ON empleos.experiencia_laboral
    IS 'Asociación entre Usuarios y Experiencia laboral';
